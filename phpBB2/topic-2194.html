S pomočjo C in OpenGL sem naredil igroc 3 v vrsto, ki lepo deluje na GNU/Linux platformi vendar ko želim to igro zagnati na računalniku z windows XP-ji mi vrne neko napako (zdele z glave nevem točno kaj napiše vendar se program ne izvede)
Če je kdo imel podoben problem in ga je rešil oz. kdo ve kako ga rešiti bi prosil za pomoč.
Program je preveden z naslednjim ukazom:
gcc -Wall -W -pedantic tictactoe.c -lm -lglut -o tic

Source code

#include <GL/glut.h>

/* Initialize variables for TicTacToe game.
*/
int player = 0;
int win = 0; /* 0 - Game not over; 1 - Draw, 2-9 - Win */
char square[3][3];
char sign = 'O';
int i, j;
float th, radius = 0.28, x, y;

void valuesToStart(void) {
player = 0;
win = 0;
for (i = 0; i < 3; i++) {
for (j = 0; j < 3; j++) {
square[i][j] = '\0';
}
}
sign = 'O';
}

void changePlayer (void) {
player++;
player %= 2;
if (player == 1) {
sign = 'X';
} else {
sign = 'O';
}
}

void testWin(void) {
if (square[0][0] != '\0' && square[1][0] != '\0' && square[2][0] != '\0'
&& square[0][1] != '\0' && square[1][1] != '\0' && square[2][1] != '\0'
&& square[0][2] != '\0' && square[1][2] != '\0' && square[2][2] != '\0') {
win = 1;
}
if (square[0][0] != '\0') {
if (square[0][0] == square[1][0]
&& square[0][0] == square[2][0]) {
win = 2;
}
if (square[0][0] == square[0][1]
&& square[0][0] == square[0][2]) {
win = 3;
}
}
if (square[1][1] != '\0') {
if (square[1][1] == square[0][1]
&& square[1][1] == square[2][1]) {
win = 4;
}
if (square[1][1] == square[0][0]
&& square[1][1] == square[2][2]) {
win = 5;
}
if (square[1][1] == square[1][0]
&& square[1][1] == square[1][2]) {
win = 6;
}
if (square[1][1] == square[2][0]
&& square[1][1] == square[0][2]) {
win = 7;
}
}
if (square[2][2] != '\0') {
if (square[2][2] == square[0][2]
&& square[2][2] == square[1][2]) {
win = 8;
}
if (square[2][2] == square[2][0]
&& square[2][2] == square[2][1]) {
win = 9;
}
}

changePlayer();
}

void reshape(int w, int h)
{
glViewport(0, 0, w, h);
glMatrixMode(GL_PROJECTION);
glLoadIdentity();
if (w <= h)
gluOrtho2D (-1.0, 1.0,
-1.0*(GLfloat)h/(GLfloat)w, 1.0*(GLfloat)h/(GLfloat)w);
else
gluOrtho2D (-1.0*(GLfloat)w/(GLfloat)h,
1.0*(GLfloat)w/(GLfloat)h, -1.0, 1.0);
glMatrixMode(GL_MODELVIEW);
glLoadIdentity();
}
