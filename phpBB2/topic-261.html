const
    naslovi:  record
        // naštejemo spremenljivke v katere bomo vpisali naslove lokacij
        A,B,C: Longint;
        // z vsebino te spremenljivke bomo prepisali vse lokacije
        E: Byte;
        // na koncu pa napolnimo spremenljivke z naslovi lokacij
    end = (A:$81CBC; B:$81CBD; C:$81CBE; E:$90;);

var
    ch: Char;
    f: File;

begin
    // povežemo s z datoteko 
    Assignfile(f,'xxx.xxx');

    // prepišemo vsebino lokacije
    Seek(f,naslovi.A);
    ch:=char(naslovi.E);
    Blockwrite(f,ch,1);

    Seek(f,naslovi.B);
    ch:=char(naslovi.E);
    Blockwrite(f,ch,1);

    Seek(f,naslovi.C);
    ch:=char(naslovi.E);
    Blockwrite(f,ch,1);

    // zapremo datoteko in postopek je končan  :D   
    CloseFile(f);
end;
