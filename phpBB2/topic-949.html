import  java.util.Vector;
import java.io.*;



/**
* Write a description of class Test here.
* testni razred za preizkus programa
* @version (25.10.2004)
*/
public class Test
{

  java.io.BufferedReader in;

    public static void meni()
    {
    System.out.println("1 vpis Osebe");
    System.out.println("2 vpis Ucenca");
    System.out.println("3 vpis Dijaka");
    System.out.println("4 vpis Studenta");
    System.out.println("5 izpis");
    System.out.println("6 IZHOD");
    }
   




    public Test() {
      //ustvarimo čitalec, ki dobi na vhodu vhodni tok podatkov iz konzole
      in = new java.io.BufferedReader(new java.io.InputStreamReader(System.in));
}

public String vnosStringa(String imevnosa) throws Exception{
    //metoda za vnos stringa z določenim imenom
    System.out.print("Vnesi "+imevnosa+":");
    return in.readLine();
    }

// zacetek glavne funkcije
public static void main(String[] a) throws Exception
{
   
    int i;
    int pomozni1;
    do
    {
   
    meni();
    Vector moj_vector =new Vector();
    Test x = new Test();
    String pomozno= x.vnosStringa("izberi zeljeno");
    i = Integer.parseInt(pomozno); //pretvorba stringa v int   


  if(i==1){
      Oseba b =new Oseba();
      b.setIme(x.vnosStringa("ime osebe "));
      b.setPriimek(x.vnosStringa("priimek osebe "));
      pomozno=(x.vnosStringa("leto rojstva"));
        pomozni1 = Integer.parseInt(pomozno); //pretvorba stringa v int
      b.setLeto(pomozni1);
      moj_vector.addElement(b);
     
}
     
     
  if(i==2){
      Ucenec c =new Ucenec();
      c.setIme(x.vnosStringa("ime ucenca "));
      c.setPriimek(x.vnosStringa("priimek ucenca "));
      c.setIme_sole(x.vnosStringa("vpisi ime sole"));
      pomozno=(x.vnosStringa("leto rojstva"));
        pomozni1 = Integer.parseInt(pomozno); //pretvorba stringa v int
      c.setLeto(pomozni1);
      moj_vector.addElement(c);
     
}
     
  if(i==3){
      Dijak c =new Dijak();
      c.setIme(x.vnosStringa("ime dijaka "));
      c.setPriimek(x.vnosStringa("priimek dijaka "));
      c.setIme_srednje_sole(x.vnosStringa("vpisi ime srednje sole"));
      c.setSmer(x.vnosStringa("vpisi smer"));
      pomozno=(x.vnosStringa("leto rojstva"));
        pomozni1 = Integer.parseInt(pomozno); //pretvorba stringa v int
      c.setLeto(pomozni1);
      moj_vector.addElement(c);
     
}
     
  if(i==4){
      Student c =new Student();
      c.setIme(x.vnosStringa("ime studenta "));
      c.setPriimek(x.vnosStringa("priimek studenta "));
      c.setSmer_studija(x.vnosStringa("vpisi smer studija"));
      pomozno=(x.vnosStringa("vpisi index"));
        pomozni1 = Integer.parseInt(pomozno); //pretvorba stringa v int
      c.setVpisna_stevilka(pomozni1);
      pomozno=(x.vnosStringa("leto rojstva"));
        pomozni1 = Integer.parseInt(pomozno); //pretvorba stringa v int
      c.setLeto(pomozni1);
      moj_vector.addElement(c);
     
}
     
  if(i==5){
     
      for (int j=0; j<moj_vector.size(); j++)
                  System.out.println((Oseba)moj_vector.elementAt(j));
                    break;  }
 

     
     
    } while (i<6);
}

   
 


   
}//konec glavnega programa
