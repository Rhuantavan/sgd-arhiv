Prepisoval sem kodo iz knjige OpenGL Game Programing, razlikuje se samo po temu da sem sprmenil nabor znakov tako da podpira unicode, zadeva se prevede a ne dela. Koda je sledeča:








Koda:
unsigned int Win32Class::NovNapis(wchar_t *ime, int velikost)
{
HFONT pisava;
unsigned int baza;

baza = glGenLists(96);

if(!_wcsicmp(ime, L"symbol"))
pisava =CreateFont(velikost, 0, 0, 0, FW_BOLD, FALSE, FALSE, FALSE,
SYMBOL_CHARSET, OUT_TT_PRECIS, CLIP_DEFAULT_PRECIS,
ANTIALIASED_QUALITY, FF_DONTCARE | DEFAULT_PITCH, ime);
else
pisava =CreateFont(velikost, 10, 10, 10, FW_BOLD, FALSE, FALSE, FALSE,
DEFAULT_CHARSET, OUT_TT_PRECIS, CLIP_DEFAULT_PRECIS,
ANTIALIASED_QUALITY, FF_DONTCARE | DEFAULT_PITCH, ime);
if(!pisava)
MessageBox(NULL, L"Pisava se ni mogle napisati", L"Opozorilo", MB_OK | MB_ICONEXCLAMATION);

SelectObject(hDC, pisava);
wglUseFontBitmaps(hDC, 32, 96, baza);

return baza;
}

void Win32Class::ZbrisiPisavo(unsigned int baza)
{
if(baza != 0)
glDeleteLists(baza, 96);
}

void Win32Class::Napisi(int x, int y, unsigned int baza, wchar_t *str)
{
if ((baza ==0) || (!str))
MessageBox(NULL, L"Tekst se ni mogle napisati", L"Opozorilo", MB_OK | MB_ICONEXCLAMATION);

glRasterPos2i(x, y);

glPushAttrib(GL_LIST_BIT);
glListBase(baza - 32);
glCallLists((int)wcslen(str), GL_UNSIGNED_BYTE, str);
glPopAttrib();
}


na začetku je še:







Koda:

potem pa :







Koda:
Napisi(100,100,64, L"pejt v rit");
