Zadnje dni me ob živce spravlja problem s projeciranjem senc (planar projected shadows, uporabljam stencil buffer). Tega ne delam prvič, zato sem skoraj živce zgubil, ker ne vem, kako naj naredim, da bo tako banalna stvar delala, kot more. Problem je, da so sence nepravilno prikazane.

Najprej koda, potem pa ugotovitve:

Tole nariše površino, kamor se senca projecira:








Koda:

///// rSurface () /////
void str_Render::rSurface (const float pos [3], const float size [3], const float rot [3], float deg)
{
glBindTexture (GL_TEXTURE_2D, floor_tex->Tex);

glPushMatrix ();

glTranslatef (pos [0], pos [1], pos [2]);
glRotatef (deg, rot [0], rot [1], rot [2]);
glScalef (size [0], size [1], size [2]);

... // nariše površino

glPopMatrix ();
}



Takole izgleda koda, ki nariše predmet, ki meče senco (ta del kode je pomemben, saj lahko vpliva na pravilnost sence):








Koda:

///// rObject () /////
/* pozornost usmerite na zadnji parameter "shadow" - ko narišemo "pravi" predmet, ima vrednost false, ko pa rišemo senco, je true - s tem množimo shadow-projection matrix
*/
void str_Render::rObject (const float pos [3], const float size [3], float rot [3], float deg, bool shadow)
{
glBindTexture (GL_TEXTURE_2D, obj_tex->Tex);

// ko bomo narisali senco...
if (shadow)
{
glPushMatrix ();
glMultMatrixf (shd_matrix);
}


glPushMatrix ();

if (shadow)
{
// pozor pri glScalef - 0.0f za y je zato, da je senca 2D
glTranslatef (pos [0], pos [1], pos [2]);
glRotatef (deg, rot [0], rot [1], rot [2]);
glScalef (size [0], 0.0f, size [2]);
}
else
{
glTranslatef (pos [0], pos [1], pos [2]);
glRotatef (deg, rot [0], rot [1], rot [2]);
glScalef (size [0],size [1], size [2]);
}

... // nariše kocko

glPopMatrix ();

if (shadow)
{
glPopMatrix ();
}
}



Transformacije vsekakor vplivajo na videz sence, saj v primeru, da ne bi klical "glScalef" s parametrom 0 za y velikost, senca sploh ne bi zgledala kot senca ampak kot razpotegnjena 3D kocka v tleh. No, sej sedaj tud ne zgleda kot senca, samo boljše je pa že.
Upam, da je ta del pravilen.

Naprej - koda, ki izriše vse skupaj:








Koda:

///// Izrisovanje scene. /////
void fncRender ()
{
glClear (GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT | GL_STENCIL_BUFFER_BIT);
glLoadIdentity ();

gluLookAt (x_p, y_p, z_p, x_l, y_l, z_l, 0.0f, 1.0f, 0.0f);

glEnable (GL_STENCIL_TEST);

glStencilFunc (GL_ALWAYS, 1, 1);
glStencilOp (GL_KEEP, GL_KEEP, GL_REPLACE);

// nariše tla v stencil buffer (področje izrisovanja) in color buffer
Render.rSurface (floor_pos, floor_size, floor_rot, 90.0f);

// nastavitve za izris sence
glEnable (GL_BLEND);
glBlendFunc (GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
glDisable (GL_DEPTH_TEST);
glDisable (GL_TEXTURE_2D);
glColor4f (0.0f, 0.0f, 0.0f, 0.5f);

glStencilFunc (GL_EQUAL, 1, 1);
glStencilOp (GL_KEEP, GL_KEEP, GL_INCR);

// izračuna projekcijsko matriko in nariše senco
// (pozorni bodite na zadnji parameter, ki je true
// - glejte funkcijo rObject ();
fncShadowMatrix (shd_matrix, plane, light_pos);
Render.rObject (cube_pos, cube_size, cube_rot, cube_degrot, true);

glColor3f (1.0f, 1.0f, 1.0f);
glDisable (GL_BLEND);
glEnable (GL_DEPTH_TEST);
glEnable (GL_TEXTURE_2D);

glDisable (GL_STENCIL_TEST);

// nariše pravi predmet - shadow = false
Render.rObject (cube_pos, cube_size, cube_rot, cube_degrot, false);
}



Kot sem omenil, senca izgleda čisto napačna. Pri prejšnjih primerih je delalo, saj nisem uporabljal glScale/Rotate/Translate, ki senco čisto razsujejo. Je mogoče tukaj kaj narobe?

Še ena možnost je, da je treba zaradi transformacij drugače izračunat matriko ali kaj podobnega. Zaenkrat izgleda takole (torej običajna in pravilna):









Koda:
void fncShadowMatrix (float shd_matrix [16], float plane [4], float light_pos [4])
{
float dot;
dot = plane [0] * light_pos [0] + plane [1] * light_pos [1] +
plane [1] * light_pos [2] + plane [3] * light_pos [3];


shd_matrix [0] = dot - light_pos [0] * plane [0];
shd_matrix [4] = 0.0f - light_pos [0] * plane [1];
shd_matrix [8] = 0.0f - light_pos [0] * plane [2];
shd_matrix [12] = 0.0f - light_pos [0] * plane [3];

shd_matrix [1] = 0.0f - light_pos [1] * plane [0];
shd_matrix [5] = dot - light_pos [1] * plane [1];
shd_matrix [9] = 0.0f - light_pos [1] * plane [2];
shd_matrix [13] = 0.0f - light_pos [1] * plane [3];

shd_matrix [2] = 0.0f - light_pos [2] * plane [0];
shd_matrix [6] = 0.0f - light_pos [2] * plane [1];
shd_matrix [10] = dot - light_pos [2] * plane [2];
shd_matrix [14] = 0.0f - light_pos [2] * plane [3];

shd_matrix [3] = 0.0f - light_pos [3] * plane [0];
shd_matrix [7] = 0.0f - light_pos [3] * plane [1];
shd_matrix [11] = 0.0f - light_pos [3] * plane [2];
shd_matrix [15] = dot - light_pos [3] * plane [3];
}



Tole pa izračuna d vrednost (iz enačbe za površino) za "plane" parameter:








Koda:
///// fncCalcDValue () /////
float fncCalcDValue (float floor_normal [3], const float floor_pos [3])
{
// A*x + B*y + C*z + D = 0;

float D;
D = floor_normal [0] * floor_pos [0] +
floor_normal [1] * floor_pos [1] +
floor_normal [2] * floor_pos [2];

return -D;
}

no kar se Shader-jev tice mislm da folk ne ve v cem je tle fora...
1.) pocak na OpenGL 2.0 oz. kup 3dlabs WildCat-a (cca. 80-100K) opengl 2.0 driverje in se ucis - k kar je trenutno na sceni je vse POMESAN... eni cudni ext in k delo le na tej kartici in svasta... se ne splaca matrat

2.) npr. DOT3 bump mapping je ze v OpenGL 1.3 ARB_texture_env_dot3
tko, da zakaj bi se tle matru s SHADERJI ce lohk tko nardis z APIji (ok vsaj po mojem mnenju je delo z apiji hitrejse, kot da gres programirat SHADERJE) -> prog. SHADERJI so namenjeni efektom, ki niso podprti s strani apijev oz. za "izdelavo" novih efektov

kar se pa tice dot3 bump-a pa:

opengl 1.3:
- ARB_texture_env_dot3
- ARB_texture_env_combine
- ARB_multitexture

pol pa:

// find dot product of N (stored in the texture map) and L (stored
// as the PRIMARY_COLOR).

glTexEnvf(GL_TEXTURE_ENV, GL_TEXTURE_ENV_MODE, GL_COMBINE);
glTexEnvf(GL_TEXTURE_ENV, GL_COMBINE_RGB, GL_DOT3_RGB);
glTexEnvf(GL_TEXTURE_ENV, GL_SOURCE0_RGB, GL_TEXTURE);
glTexEnvf(GL_TEXTURE_ENV, GL_OPERAND0_RGB, GL_SRC_COLOR);
glTexEnvf(GL_TEXTURE_ENV, GL_SOURCE1_RGB, GL_PRIMARY_COLOR);
glTexEnvf(GL_TEXTURE_ENV, GL_OPERAND1_RGB, GL_SRC_COLOR);

// modulate the base texture by N.L

glTexEnvf(GL_TEXTURE_ENV, GL_TEXTURE_ENV_MODE, GL_COMBINE);
glTexEnvf(GL_TEXTURE_ENV, GL_COMBINE_RGB, GL_MODULATE);
glTexEnvf(GL_TEXTURE_ENV, GL_SOURCE0_RGB, GL_PREVIOUS);
glTexEnvf(GL_TEXTURE_ENV, GL_OPERAND0_RGB, GL_SRC_COLOR);
glTexEnvf(GL_TEXTURE_ENV, GL_SOURCE1_RGB, GL_TEXTURE);
glTexEnvf(GL_TEXTURE_ENV, GL_OPERAND1_RGB, GL_SRC_COLOR);

L (stored as the PRIMARY_COLOR) -> normaliziran light vector (za vsak trikotnik po 3 vektorji, torej za vsako tocko trikotnika izracunas lightvector in ga zapises v barvo trenutne tocke - potem se zadeva interpolira in dobis "Per-Pixel Lighting" in to je to - ce renderiras z texturami mas dot3 bump drugac pa Per-Pixel Lighting renderirano sceno = a pa ta zadeva dela povsod ok tm od GeForce >)
