Na netu se svaljka koda za loadanje Doom3 modelov...

še koda (če koga zanima).









Koda:
#include <cstdio>
#include <cstring>
#include <string>
#include <cmath>
using namespace std;

struct Bone
{
char name[64];
float pos[3];
float rot[9];
int parent;
};

struct Vertex
{
float u, v;
int weight_index;
int weight_count;
float x, y, z;
float nx, ny, nz;
};

struct Triangle
{
int a, b, c;
};

struct Weight
{
int bone;
float t;
float x, y, z;
};

struct Mesh
{
char texture[512];
int nb_vertices;
Vertex* vertices;
int nb_triangles;
Triangle* triangles;
int nb_weights;
Weight* weights;
};

char g_CurrentLine[1024];
int g_NbMeshes = 0;
Mesh* g_Meshes = 0;
int g_NbBones = 0;
Bone* g_Bones;

bool ReadLine(FILE* fp)
{
int pos = 0;

while (1)
{
int inchar = fgetc(fp);

if (inchar == EOF)
{
g_CurrentLine[pos] = 0;
return (false);
}

if (inchar == 13)
{
fgetc(fp);
break;
}

g_CurrentLine[pos++] = inchar;
}

g_CurrentLine[pos] = 0;

return (true);
}

bool ReadDOOMFile(const char* filename)
{
FILE* fp = fopen(filename, "rb");
if (fp == 0)
return (false);

// Skip header
ReadLine(fp);
ReadLine(fp);

// ========== BONES ==========
ReadLine(fp);
if (!memcmp(g_CurrentLine, "numbones", 8))
{
sscanf(g_CurrentLine, "numbones %d", &g_NbBones);
g_Bones = new Bone[g_NbBones];

printf("Number of bones: %d\n", g_NbBones);

for (int i = 0; i < g_NbBones; i++)
{
Bone& bone = g_Bones[i];

// Start of the bone
ReadLine(fp);

// ========== NAME ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tname \"%s", bone.name);
bone.name[strlen(bone.name) - 1] = 0;

// ========== POSITION ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tbindpos %f %f %f", &bone.pos[0], &bone.pos[1], &bone.pos[2]);

// ========== ROTATION ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tbindmat %f %f %f %f %f %f %f %f %f",
&bone.rot[0],
&bone.rot[1],
&bone.rot[2],
&bone.rot[3],
&bone.rot[4],
&bone.rot[5],
&bone.rot[6],
&bone.rot[7],
&bone.rot[8]);

// ========== PARENT ==========
ReadLine(fp);
bone.parent = -1;
if (g_CurrentLine[0] != '}')
{
char parent[64];
sscanf(g_CurrentLine, "\tparent \"%s", parent);
parent[strlen(parent) - 1] = 0;
for (int j = 0; j < i; j++)
{
if (!strcmp(parent, g_Bones[j].name))
{
bone.parent = j;
break;
}
}
ReadLine(fp);
}

// Separating line
ReadLine(fp);
}
}

// ========== MESHES ==========
ReadLine(fp);
if (!memcmp(g_CurrentLine, "nummeshes", 9))
{
sscanf(g_CurrentLine, "nummeshes %d", &g_NbMeshes);
g_Meshes = new Mesh[g_NbMeshes];

printf("Number of meshes: %d\n", g_NbMeshes);

ReadLine(fp);

for (int i = 0; i < g_NbMeshes; i++)
{
printf("Mesh %d\n", i);

Mesh& mesh = g_Meshes[i];

// Start of mesh
ReadLine(fp);

// ========== SHADER ==========
ReadLine(fp);
char* ptr = g_CurrentLine;
while (*ptr != '\"')
ptr++;
strcpy(mesh.texture, ptr + 1);
mesh.texture[strlen(mesh.texture) - 1] = 0;
ReadLine(fp);
printf(" Shader: %s\n", mesh.texture);

// ========== VERTICES ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tnumverts %d", &mesh.nb_vertices);
printf(" Number of vertices: %d\n", mesh.nb_vertices);
mesh.vertices = new Vertex[mesh.nb_vertices];
for (int j = 0; j < mesh.nb_vertices; j++)
{
int index;
Vertex& vertex = mesh.vertices[j];
ReadLine(fp);
sscanf(g_CurrentLine, "\tvert %d %f %f %d %d",
&index,
&vertex.u,
&vertex.v,
&vertex.weight_index,
&vertex.weight_count);
vertex.v = 1 - vertex.v;
}
ReadLine(fp);

// ========== TRIANGLES ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tnumtris %d", &mesh.nb_triangles);
printf(" Number of triangles: %d\n", mesh.nb_triangles);
mesh.triangles = new Triangle[mesh.nb_triangles];
for (int j = 0; j < mesh.nb_triangles; j++)
{
int index;
Triangle& triangle = mesh.triangles[j];
ReadLine(fp);
sscanf(g_CurrentLine, "\ttri %d %d %d %d",
&index,
&triangle.a,
&triangle.b,
&triangle.c);
}
ReadLine(fp);

// ========== WEIGHTS ==========
ReadLine(fp);
sscanf(g_CurrentLine, "\tnumweights %d", &mesh.nb_weights);
printf(" Number of weights: %d\n", mesh.nb_weights);
mesh.weights = new Weight[mesh.nb_weights];
for (int j = 0; j < mesh.nb_weights; j++)
{
int index;
Weight& weight = mesh.weights[j];
ReadLine(fp);
sscanf(g_CurrentLine, "\tweight %d %d %f %f %f %f",
&index,
&weight.bone,
&weight.t,
&weight.x,
&weight.y,
&weight.z);
}
ReadLine(fp);

ReadLine(fp);
}
}

fclose(fp);

return (true);
}

void BuildVertices(void)
{
for (int i = 0; i < g_NbMeshes; i++)
{
Mesh& mesh = g_Meshes[i];

for (int j = 0; j < mesh.nb_vertices; j++)
{
Vertex& v = mesh.vertices[j];

v.x = 0;
v.y = 0;
v.z = 0;

for (int k = 0; k < v.weight_count; k++)
{
Weight& w = mesh.weights[v.weight_index + k];

Bone& b = g_Bones[w.bone];

v.x += (b.rot[0] * w.x + b.rot[3] * w.y + b.rot[6] * w.z + b.pos[0]) * w.t;
v.z += (b.rot[1] * w.x + b.rot[4] * w.y + b.rot[7] * w.z + b.pos[1]) * w.t;
v.y += (b.rot[2] * w.x + b.rot[5] * w.y + b.rot[8] * w.z + b.pos[2]) * w.t;
}
}
}
}

void BuildNormals(void)
{
for (int i = 0; i < g_NbMeshes; i++)
{
Mesh& mesh = g_Meshes[i];

for (int j = 0; j < mesh.nb_vertices; j++)
{
mesh.vertices[j].nx = 0;
mesh.vertices[j].ny = 0;
mesh.vertices[j].nz = 0;
}

for (int j = 0; j < mesh.nb_triangles; j++)
{
Triangle& tri = mesh.triangles[j];

Vertex& v0 = mesh.vertices[tri.a];
Vertex& v1 = mesh.vertices[tri.b];
Vertex& v2 = mesh.vertices[tri.c];

float Ax = v1.x - v0.x;
float Ay = v1.y - v0.y;
float Az = v1.z - v0.z;

float Bx = v2.x - v0.x;
float By = v2.y - v0.y;
float Bz = v2.z - v0.z;

float nx = Ay * Bz - By * Az;
float ny = -(Ax * Bz - Bx * Az);
float nz = Ax * By - Bx * Ay;

v0.nx += nx;
v0.ny += ny;
v0.nz += nz;
v1.nx += nx;
v1.ny += ny;
v1.nz += nz;
v2.nx += nx;
v2.ny += ny;
v2.nz += nz;
}

for (int j = 0; j < mesh.nb_vertices; j++)
{
Vertex& v = mesh.vertices[j];

float len = sqrt(v.nx * v.nx + v.ny * v.ny + v.nz * v.nz);

if (len)
{
v.nx /= len;
v.ny /= len;
v.nz /= len;
}
}
}
}

int main(int argc, const char* argv[])
{
if (ReadDOOMFile(argv[1]))
{
BuildVertices();
BuildNormals();

// Insert your engine exporter here
}

return (0);
}
