KAJ JE COM?
COM (component object model) je M$-jev standard (delujoc samo v win32) za uporabo objektov implementiranih v knjiznicah v programih preko interfacov. Zakajje ql? Uporaba objektnega pristopa (kar je trenutno zadnji hit mode) & vmesni layer med objektom (knjiznico) in programom. Ta vmesna plast zagotavlja, da se objekt lahko uporablja ne da bi imel uporabnik dostop do njegovega privatnega pomnilnika in kode (kar je zelo prakticno za komercialne knjiznice aka DX, ki ne zelijo razkriti svoje vsebine razvijalcu, a kljub temu biti objektne).

"FREE COM"?
Zaradi neprenosljivosti COM-a, so se pojavile free implementacije le tega, ki so ~95% kompatibilne s COM (ideja in nacin dela sta enaka).

PRILOZENA KODA:
Je pisana za linux (je bil ravno pri roki, teoreticno naj bi se koda kompilirala tudi v win32 pod ming32 ali pa VC, saj ne vsebuje nobenih posebnih sys klicev). Sestavljato jo knjiznica in majhen programcic, ki uporablja le to. Knjiznica v sebi vsebuje objekt CDog (pac prvo kar mi je padlo na pamet) do katerega program dostopa preko IDog. Da dobi prvo referenco mora poklicati CreateIDog() (funkcjo, ki jo DLL eksporta), kar je podobno kot pri DX (tam npr. je potrebno na zacetku dobiti interface s klicem CreateIDirect3D9() ). Nadalje pa delamo preko interface.
Kako kompilirati: "g++ -shared libct.cpp -o libct.so\n g++ ct.cpp ./libct.so -o ct\n ./ct"!

OPOZORILO:
Razlaga, koda etc. lahko vsebije strokovne in slovnicne napake!

NADALJE:
Sedaj prosim povejte, kaj naj se napisem, ker pac ne vem, kaj vam se ni jasno in kaj vas se zanima!









Koda:

//ct.h - shared header


#define IID_IDOG 1
#define IID_IDOGNAME 2
#define IID_IDOGHEAD 3


class IUnknown
{
public:
virtual int AddRef() = 0;
virtual int Release() = 0;
virtual void* QueryInterface(int IID) = 0;
};


class IDog : public IUnknown
{
public:
virtual void SayHi() = 0;
};


class IDogName : public IUnknown
{
public:
virtual void Ask() = 0;
};


class IDogHead : public IUnknown
{
public:
virtual void Save(int) = 0;
virtual int Load() = 0;
};


IDog* CreateIDog();













Koda:

//libct.cpp - library

#include <stdio.h>

#include "ct.h" //include-an standardni header


class CDog : public IDog, IDogName, IDogHead //class mora podedovati vse interface, ki jih zeli kasneje izraziti
{
private:
int refcount; //notranji stevec referenc
int dogmemory; //poljubne spremenljivke za zasebno rabo...
public:
//definicija standardnih funkciji iz IUnknown
virtual int AddRef();
virtual int Release();
virtual void* QueryInterface(int);

//poljuben konstruktor in destruktor
CDog();
~CDog();

//funkcije tega razreda...
virtual void SayHi();
virtual void Ask();
virtual void Save(int);
virtual int Load();
};


int CDog::AddRef() //doda eno referenco
{
refcount++;
return refcount;
}


int CDog::Release() //odvzame eno referenco in ce objekt vec ni potreben poklice destruktor
{
refcount--;
if (refcount <= 0)
delete this;
return refcount;
}


void* CDog::QueryInterface(int IID) //vrne druge mozne interface do istega objekta
{
switch (IID)
{
case IID_IDOG:
AddRef();
return (IDog*)this;
case IID_IDOGNAME:
AddRef();
return (IDogName*)this;
case IID_IDOGHEAD:
AddRef();
return (IDogHead*)this;
default:
return 0;
}
}


CDog::CDog()
{
printf("constructor...\n");
refcount = 1; //zacetno stevilo referenc je logicno na zacetku enako 1
dogmemory = 0;
}


CDog::~CDog()
{
printf("destructor...\n");
dogmemory = 0;
}


void CDog::SayHi()
{
printf("DOGGY: Hi!\n");
}


void CDog::Ask()
{
printf("DOGGY: My name is Fifi!\n");
}


void CDog::Save(int val)
{
dogmemory = val;
}


int CDog::Load()
{
return dogmemory;
}


IDog* CreateIDog() //osnovna funkcjia, ki mora biti eksportana iz knjiznice saj predstavlja osnovno vhodno rution, da sploh pridobimo prvo referenco
{
CDog* dog = new CDog(); //ustavri class (viden samo znotraj knjiznice)
IDog* idog = (IDog*)dog; //class pretvori s pomocjo typecasta v interface, ki bo viden uporabniku
return idog;
}














Koda:

//ct.cpp - user application

#include <stdio.h>

#include "ct.h"


int main()
{
printf("CTUA\n");

IDog* doggy;
doggy = CreateIDog(); //query-amo prvi osnoven interface (s pomocjo edine funkcje, ki nam jo ponuja knjiznica)

doggy->SayHi(); //ga uporabimo...

IDogName* dogname;
dogname = (IDogName*)doggy->QueryInterface(IID_IDOGNAME); //si zazelimo dodatne zmogljivosti, ki jih ponuja drugi interface

dogname->Ask();
dogname->Release(); //ga vec ne potrebujemo in ga zato release-amo

IDogHead* doghead;
doghead = (IDogHead*)doggy->QueryInterface(IID_IDOGHEAD); //prvotni interface zaprosimo se za en drug

doghead->Save(123);
doghead->Release(); //tudi tega sprostimo

doghead = (IDogHead*)doggy->QueryInterface(IID_IDOGHEAD); //in ga ponovno zahtevamo...

int x = doghead->Load();
printf("doghead: x=%i\n", x);

doggy->Release(); //sprostimo osnovni interface (in ob tem ostali se vedno ostanejo)

doghead->Save(321);
x = doghead->Load();
printf("doghead: x=%i\n", x);
doghead->Release(); //sprostimo se zadnji interface - sprozi se destruktor

printf("DONE!\n");
return 0;
}
