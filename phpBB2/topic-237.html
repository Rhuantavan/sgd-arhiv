Uspelo mi je uspsoobit SDl, jupiiii.

Ok imam tole kodo :









Koda:

#include <SDL.h>

enum {

SCREENWIDTH = 500,
SCREENHEIGHT = 500,
SCREENBPP = 0,
SCREENFLAGS = SDL_ANYFORMAT
};

int main ( int argc, char* argv[] ) {

SDL_Init ( SDL_INIT_VIDEO );
atexit ( SDL_Quit );

SDL_Surface* pSurface;
pSurface = SDL_SetVideoMode ( SCREENWIDTH, SCREENHEIGHT, SCREENBPP, SCREENFLAGS );
SDL_FillRect ( pSurface , NULL , SDL_MapRGB ( pSurface->format , 0 , 255 , 0 ) ) ;
SDL_EnableKeyRepeat( 10, 10 );

SDL_Rect kvadrat;
kvadrat.x = 50;
kvadrat.y = 50;
kvadrat.w = 50;
kvadrat.h = 50;

SDL_FillRect ( pSurface , &kvadrat , SDL_MapRGB ( pSurface->format , 255, 0, 0 ) );
SDL_UpdateRect ( pSurface , 0 , 0 , 0 , 0 );

SDL_Event event;
while ( true ) {

if ( SDL_PollEvent ( &event) ) {

if ( event.type == SDL_QUIT ) break;
else if ( event.type == SDL_KEYDOWN ) {

if ( event.key.keysym.sym == SDLK_UP ) {

kvadrat.y = kvadrat.y - 5;
}
else if ( event.key.keysym.sym == SDLK_DOWN ) {

kvadrat.y = kvadrat.y + 5;
}
else if ( event.key.keysym.sym == SDLK_LEFT ) {

kvadrat.x = kvadrat.x - 5;
}
else if ( event.key.keysym.sym == SDLK_RIGHT ) {

kvadrat.x = kvadrat.x + 5;
}
SDL_FillRect ( pSurface , &kvadrat , SDL_MapRGB ( pSurface->format , 255, 0, 0 ) );
SDL_UpdateRect ( pSurface, 0, 0, 0, 0 );
}
}


}


return 0;
}

Hja, kako narediti da bo izginjal?

Imas kar nekaj opcij:

Prva logicna je SDL alpha blending ... narises vec kvadratkov, tolikor, kolikor je dolg "rep" in vsakega pac z drugacno prozornostjo "alpha". Za to imas func SDL_MapRGBA, mislim pa da morasv nekaj enablat da to dela

Druga bi lahko bila taka, kot se uporablja za generacijo dejanskih tekstur ognja in plazme (torej da upostevas da ima vsak pixel temparaturo, ki z casom zbledi, je pa res, da vpliva tudi na sosednje pixle)

Tretja bi bila pa recimo, da bi si spisal en specificni draw rect, da bi lerpal po repu barvo v MapRGB(0,0,0)
