The Draw pointer-to-member is defined as a member variable, but is then used as if it were a non-member. Correct syntax would have been (turret.*(turret.Draw))(angle);

Using typedefs for the function types would have been a good idea.

So would have been pointing out that you cannot manipulate bound pointers-to-members (particularly, you can't store them).

To se mi zdi skrajno cuden pristop. To se da s C++ lepo resit, samo je malo bolj obsezno. Lahko pa tak razmisljas:

npr. da imas button, edit in scroll kontrole. Vsi tile so derivati od nekega objekta, ki je torej vsem skupen, in ta objekt se lahko poda slikarju. En pristop je tale:

class CDescriptor
{
protected:
int x,y,w,h;
std::string caption;
}

class CPainter
{
public:
virtual void Draw( CDescriptor* pObjekt );
};

class CWidget : public CDescriptor;
{
protected:
CPainter* m_pPainter;
public:

CWidget()
{
m_pPainter = NULL;
}

inline void SetPainter( CPainter* pPainter ) { m_pPaint = pPainter; }

virtual void OwnPaint()
{
}

void Paint()
{
if( m_pPainter != NULL ) m_pPainter->Draw( this ); else OwnPaint();
}

}

class CButton : public CWidget
class CEdit : public CWidget
class CScroll : public CWidget

tako , zdaj pa ce implementiras CButton in mu v OwnPaint reces kako naj se narise je vse ok. Ce pa implementiras CPainter pod-objekt pa lahko naredis takole:

CButton gumbek;
gumbek.Paint(); //default
CRdeciPainter red;
gumbek.SetPainter( &red );
gumbek.Paint(); //red skin
