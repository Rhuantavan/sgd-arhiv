Možnosti je veliko. Lahko si recimo zapomniš nek referenčni čas na začetku izrisa, na koncu pa izvedeš sleep za preostanek časa (kar seveda izračunaš).

V vsakem primeru pa se ti po mojem splača prilagoditi izris tako, da upošteva dejanski pretečen čas.

Še nekaj sem se spomnil. Glede na to da te zanimajo animacije zna biti odgovor naslednji. Če imaš fixed rate animacije in jih teče več naenkrat (z različnimi rate-i), potem najbolje da izrisuješ vsak frame. Pri vsaki tekoči animaciji (zapakirani v razredu) pa hraniš čas ki mora preteči do naslednjega frame-a (lahko tudi absolutni čas kdaj je treba spremeniti frame). Vsak frame preveriš če je ta čas že potekel in v tem primeru greš na naslednji frame (in seveda nastaviš nov čas za naslednji animation frame, popravljen za morebitno prekasno zamenjavo trenutnega framea).

/******************************************************************************
File Name: AXP_Timer.h
Class Name: AXP_Timer
Version: 1.0.2 (major.minor.build)
Author: Marko Ravnjak a.k.a. Vesoljc
Author Email: vesoljc@hotmail.com
Description: AXP_Timer class uses multimedia performace timer to
keep track of the system time.
Usage: - Initialize()
- loop Update()
Updates: 26.09.2002 - file created
******************************************************************************/

#ifndef AXP_TIMER_H
#define AXP_TIMER_H

#include "axpConstants.h"
#include "axpCLL.h"

class AXP_Timer: public axpID
{
private:
// TIMES
double m_t_factor; // time factor
double m_t_elapsed; // elapsed time
LONGLONG m_t_last; // last time
LONGLONG m_t_current; // current time
LONGLONG m_t_performance_counter; // performace counter
double m_t_total; // total time

// FRAMES
LONGLONG m_f_total; // total frames
double m_f_average; // average fps
double m_f_last_sec; // fps in last sec

// List of last second elapsed times
AXP_CLL<double>* m_times; // list of elapsed times
double m_t_list_total; // total time of list

bool m_active;

// Misc functions
void FixList ();
double GetListTotalTime ();

public:
AXP_Timer (char* name);
virtual ~AXP_Timer ();

bool Initialize ();
void Destroy ();

void Log ();

// DATA Control
void Update ();

LONGLONG GetCurrentTime ();
double GetElapsedTime ();
double GetTotalTime ();
void SetLastTime (LONGLONG in);
double GetFPS ();
double GetAFPS ();
double GetTimeFactor ();
LONGLONG GetTotalFrames ();
};

#endif // AXP_TIMER_H











Koda:

/******************************************************************************
File Name: AXP_Timer.cpp
Class Name: AXP_Timer
Author: Marko Ravnjak a.k.a. Vesoljc
Author Email: vesoljc@hotmail.com
******************************************************************************/

#include "axpTimer.h"

///////////////////////////////////////////////////////////////////////////////
// CONSTRUCTION / DESTRUCTION //
///////////////////////////////////////////////////////////////////////////////

// AXP_Timer()
// Basic Constructor
AXP_Timer::AXP_Timer(char* name): axpID(name)
{
m_t_factor = 0;
m_t_elapsed = 0;
m_t_last = 0;
m_t_current = 0;
m_t_performance_counter = 0;
m_t_total = 0;
m_f_total = 0;
m_f_average = 0;
m_f_last_sec = 0;
m_t_list_total = 0;

m_times = NULL;
}

// AXP_Timer()
// Basic Destructor
AXP_Timer::~AXP_Timer()
{
if(m_times != NULL)
m_times->Destroy();
SafeDelete(m_times)
}

///////////////////////////////////////////////////////////////////////////////
// PRIVATE MEMBERS //
///////////////////////////////////////////////////////////////////////////////

// void FixList()
// FixList function is removing elapsed times from the list so long, that
// list total time is under one second
void AXP_Timer::FixList()
{
double temp;
if(m_times != NULL)
{
temp = GetListTotalTime();
while(temp > 1.0f)
{
m_times->ResetWalker();
m_times->WalkForward();
m_times->DeleteData(m_times->GetWalkerIdent());
temp = GetListTotalTime();
}
}
}

// double GetListTotalTime()
// Calculates list total time
double AXP_Timer::GetListTotalTime()
{
double temp = 0;
if(m_times != NULL)
{
m_times->ResetWalker();
for(UINT i=0;i<(m_times->GetCounter());i++)
{
m_times->WalkForward();
temp += m_times->GetWalkerData();
}
m_t_list_total = temp;
}
return(temp);
}

///////////////////////////////////////////////////////////////////////////////
// PUBLIC MEMBERS //
///////////////////////////////////////////////////////////////////////////////

// bool Initialize()
// Initializes AXP_Timer
bool AXP_Timer::Initialize()
{
if(QueryPerformanceFrequency((LARGE_INTEGER *) &m_t_current) == 0)
{
return(false);
}

m_t_factor = 1.0f / m_t_current;
QueryPerformanceCounter((LARGE_INTEGER *) &m_t_last);
m_active = true;

m_times = new AXP_CLL<double>("m_times");
if(m_times == NULL)
{
return(false);
}

if(!m_times->Initialize())
return(false);
return(true);
}

// void Destroy()
// Destroys AXP_Timer
void AXP_Timer::Destroy()
{
if(m_times != NULL)
m_times->Destroy();
SafeDelete(m_times)
}

// void Update()
// Updates AXP_Timer, gets new times and calculates frames
void AXP_Timer::Update()
{
if(m_active)
{
QueryPerformanceCounter((LARGE_INTEGER *) &m_t_current);
m_t_elapsed = (m_t_current - m_t_last) * m_t_factor;

m_f_total++;
m_t_total+= m_t_elapsed;

m_times->AddData(m_t_elapsed,NULL);
FixList();
m_f_last_sec = (m_times->GetCounter()) / m_t_list_total;
m_f_average = double(m_f_total / m_t_total);
}
}

// LONGLONG GetCurrentTime()
// Returns current time
LONGLONG AXP_Timer::GetCurrentTime()
{
return(m_t_current);
}

// double GetElapsedTime()
// Returns elapsed time
double AXP_Timer::GetElapsedTime()
{
return(m_t_elapsed);
}

// double GetTotalTime
// Returns total time (since app started)
double AXP_Timer::GetTotalTime()
{
return(m_t_total);
}

// void SetLastTime(LONGLONG in)
// Sets new last time
void AXP_Timer::SetLastTime(LONGLONG in)
{
m_t_last = in;
}

// double GetFPS()
// Returns Frames Per Second in the last sec
double AXP_Timer::GetFPS()
{
return(m_f_last_sec);
}

// double GetAPFS()
// Returns Average FPS (since the app started)
double AXP_Timer::GetAFPS()
{
return(m_f_average);
}

// void Log()
// Logs subsystem data
void AXP_Timer::Log()
{
gLog.Info("<br><b>AXP_Timer '%s' Log:</b>",Ident().data());
gLog.Info("<li>Total Time = '%.10f sec'",m_t_total);
gLog.Info("<li>Current Time = '%d'",m_t_current);
gLog.Info("<li>Elapsed Time = '%.10f sec'",m_t_elapsed);
gLog.Info("<li>Last Time = '%d'",m_t_last);
gLog.Info("<li>Time Factor = '%.10f'",m_t_factor);
gLog.Info("<li>Total Frames = '%d'",m_f_total);
gLog.Info("<li>Average Frames = '%.2f'",m_f_average);
gLog.Info("<li>Last Sec Frames = '%.2f'",m_f_last_sec);
}

// double GetTimeFactor()
// Returns time factor
double AXP_Timer::GetTimeFactor()
{
return(m_t_factor);
}

// LONGLONG GetTotalFrames()
// Returns total frames
LONGLONG AXP_Timer::GetTotalFrames()
{
return(m_f_total);
}

// VIRTUAL MEMBERS //
