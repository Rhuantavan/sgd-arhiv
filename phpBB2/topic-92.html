No, pa ti vseeno en kos kode pejstam, da vidm ce deluje:









Koda:

Public Function TestLine(ByVal ps As PointF, ByVal pe As PointF, ByVal pathpoints() As PointF) As Boolean
Dim a As Integer
Dim result As Boolean
Dim p1, p2 As Single
For a = 0 To pathpoints.Length - 1
'test triangle a,a+1,a+2
Dim i(1) As Integer
i(0) = a
i(1) = a + 1
If i(1) > pathpoints.Length - 1 Then i(1) -= pathpoints.Length
result = LineIntersect(ps, pe, pathpoints(i(0)), pathpoints(i(1)), p1, p2)
If result = True Then
If Not ((p1 = 0 Or p1 = 1) And (p2 = 0 Or p2 = 1)) Then
Return False
End If
End If
Next
Dim pm As PointF = New PointF((pe.X - ps.X) / 2 + ps.X, (pe.Y - ps.Y) / 2 + ps.Y)
Dim polygon(pathpoints.Length - 1) As PointF
For a = 0 To pathpoints.Length - 1
polygon(a).X = pathpoints(a).X
polygon(a).Y = pathpoints(a).Y
Next
Return PointInPoly(pm, polygon)

End Function
