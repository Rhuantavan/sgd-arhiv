Koda:

#include <iostream>

class CNode {
public:
CNode* next;

CNode() : next(0) {}
virtual ~CNode() {
if (next) delete next;
}

virtual void Print() = 0;
};


class CNodeLong : public CNode {
long data;
public:
CNodeLong(long newData) : data(newData) {}
void Print() {
std::cout << data;
}
};


class CNodeString : public CNode {
char* data;
public:
CNodeString(char* newData) : data(newData) {}
void Print() {
std::cout << data;
}
};


class CLinearList {
CNode* top;
public:
CLinearList() : top(0) {}
~CLinearList() { Clean(); }

void Add(CNode* newNode) {
CNode *p = top;
while (p) {
p = p->next;
}
p = newNode;
}

CNode* Get() {
if(top == 0) return 0;
CNode* p = top;
CNode* p2 = top;
top = top->next;
delete p;
return p2;
}

void Print() {
CNode *p = top;
while (p) {
p->Print();
std::cout << " ";
p = p->next;
}
}

private:
void Clean() {
CNode *p = top;
while (p) {
top = top->next;
delete p;
p = top;
}
}
};


int main() {
CNodeLong *nlong = new CNodeLong(1323421);
CNodeString *nstring = new CNodeString("sfasfas");
CNodeString *nstring2 = new CNodeString("gpsijgd");

CLinearList list;
list.Add(nlong);
list.Add(nstring);
list.Add(nstring2);
list.Print();
list.Get();
list.Print();
return 0;
}



Dolgo se nisem s C++ ukvarjal in stvari že malo pozabljam. Je komu jasno kako naj napišem metodo Get(), ki naj vrne prvo vrednost iz seznama in jo tudi izbriše? Enostavno bi bilo v primeru, da bi imel podatek (node) le enega tipa, ampak v mojem primeru, ko je lahko podatek bodisi tipa CNodeLong , kot tudi CNodeString je stvar malo bolj kompleksna... virtualne metode.
