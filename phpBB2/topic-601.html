// Server z funkcijama send() & recv()

#include <winsock2.h> // uporabmo za povezavo prek neta
#include <stdio.h> // za I/O

// startupServerForListning() nam vrne socket ki so boundane na port ki smo ga dolocili
// ce bo kaksna napaka nam vrne -1

int startupServerForListning(unsigned short port);

// shutdown() bo izklopilo socket in izklopilo program ki smo ga prizgali

void shutdownServer(int socket);

void main()

{
printf(" Server V 0.1 v izdelavi :) mogoe bo chat prog mogoce trojan kaj te ves\n");

// initializiramo spremenljivko serversocket

int serverSocket;

// zazenemo server z to funkcijo

serverSocket = startupServerForListning(7777);

// preverimo error

if ( serverSocket == -1)
{
printf(" Startup spodletel! \n Program se zakljucuje\n ");
return;
}

// sprejmemo client

int clientSocket;

clientSocket = accept(serverSocket, 0, 0);

// preverimo error

if (clientSocket == SOCKET_ERROR)
{
printf(" Sprejemanje spodletelo!\n");
}

// inicializiramo koliko bajtov dovolimo da se rpenese

int nBytes;

// sedaj rabimo nekaj bufferjev , rabimo ga za posiljat podatke in prostor kam jih bomo shranjevali
// spodaj definiramo koliko bajtov lahko sprejmemo in poslemo

#define MESSAGE_SIZE 24

// in se buffer

char inMessage[MESSAGE_SIZE];
char outMessage[MESSAGE_SIZE] = " To posilja serv.";

// ko se bo client povezal bo poslal sporocilo serveru server pa nazaj sporocilo clientu
// spodaj uporabimo funk. recv() in flag 0 ki ga ne razumem neke prevec :)
// nato se preverimo ali je kaksen error ali ne

nBytes = recv(clientSocket, inMessage, sizeof(inMessage), 0);

// preverimo error

if (nBytes == SOCKET_ERROR)
{
printf("Sprejemanje spodletelo!\n");
}

// na tej tocki smo sprejeli podatke in shranli v inMessage
// zato izpisemo buffer da vidimo kaksno je bilo sporocilo

printf("Sporocilo prispelo: %s\n", inMessage);

// sedaj smo sprejeli sporocilo zdaj pa se moremo poslati clientu nekaj nazaj z funkcijo send()
// funkcija send() ima 4 parametre kot funk. recv()...
// posljemo sizeof(outMessage) k klientu

nBytes = send(clientSocket, outMessage, sizeof(outMessage), 0);

// preverimo error

if (nBytes == SOCKET_ERROR)
{
printf(" Sporocilo spodletelo!\n");
}
else
{
printf(" Sporocilo dostavljeno\n");
}

// sedaj nam je uspelo poslati in sprejeti sporocilo
// sedaj ko nam je uspelo sprejeti in poslati moramo se za sabo pocistiti
// zato zapremo socket

closesocket(clientSocket);

// in izklopimo povezavo

shutdownServer(serverSocket);

printf(" Pritisni tipko za nadaljevanje ...\n");
getchar();

}

//--------------------------------------------------------------------
// startupServerForListning() - funkcija ki zazene winsock in odpre socket da cakajo na klic

int startupServerForListning(unsigned short port)

{
// winsock podatkovna struktura

WSAData wsaData;

// zazenemo winsock

if (WSAStartup(MAKEWORD(2, 2), &wsaData) == SOCKET_ERROR)
{
printf(" Ne morem zagnati winsock \n");
return -1;
}

// napravimo mysocket

int mySocket = socket(AF_INET, SOCK_STREAM, 0);

// preverimo errore

if (mySocket == SOCKET_ERROR)
{
printf(" Ne gre odpret socket!\n");
return -1;
}

// struktura o naslovu

struct sockaddr_in server;

// napolnimo strukturo z podatki

server.sin_family = AF_INET;
server.sin_port = htons(port);
server.sin_addr.s_addr = INADDR_ANY;

// sedaj bindamo socket

if (bind(mySocket, (sockaddr*)&server, sizeof(server)) == SOCKET_ERROR)
{
printf(" Bindanje ni uspelo!\n");
closesocket(mySocket);
return -1;
}

// pripravimo socket da cakajo na klic

if (listen(mySocket, 5) == SOCKET_ERROR)
{
printf(" Cakanje spodletelo!\n");
closesocket(mySocket);
return -1;
}

printf(" Server zagnan\n");

return mySocket;
}

/*--------------------------------------------------------------
shutdownServer() - funkcija za zapret socket in ugasnit povezavo*/

void shutdownServer(int socket)

{

// zapremo socket

closesocket(socket);

// ugasnemo winsock

WSACleanup();

printf(" Server izkloplen\n");

}

// Client z funkcijama send() & recive()
// startup funkcija se bo konektala na dolocen server in vrnila -1 ce bo kje kaksen error

int startupClient(unsigned short port, const char* serverName);

// shutdownClient() bo zaprla socket in ugasnila winsock

void shutdownClient(int clientSocket);


#include <winsock2.h> // winsock za povezavo prek interneta
#include <stdio.h> // za I/O

void main()

{
printf(" Toturial o clientu \n ");

int mySocket;

// zazenemo client in se probamo povezat

mySocket = startupClient(7777, "localhost");

if (mySocket == -1)
{
// naslo je error izklopimo klienta in adio :)
shutdownClient(mySocket);
return;
}

// naredimo spremenljivko nBytes ta bo dovolila koliko bajtov bo lahko slo do serverja prek enega sporocila

int nBytes;

// tu definiramo koliko jih bo slo skozi

#define MESSAGE_SIZE 24

// in buffers

char inMessage[MESSAGE_SIZE];
char outMessage[MESSAGE_SIZE] = " To send iz client";

// pošljimo sporocilo serveru

nBytes = send(mySocket, outMessage, sizeof(outMessage), 0);

// preverimo error

if (nBytes == SOCKET_ERROR)
{
printf(" Posiljanje spodletelo! \n");
}
else
{
// sprinta sporocilo ki ga je poslalo
printf("Sporocilo: %s\n", outMessage);
}

// tu pricakujemo nekaj nazaj od serverja

nBytes = recv(mySocket, inMessage, sizeof(inMessage), 0);

// preverimo errore

if (nBytes == SOCKET_ERROR)
{
printf(" Sprejemanje spodletelo!\n");
}
else
{
printf("Sporocilo dostavljeno: %s\n", inMessage);
}

// zaprimo clienta
shutdownClient(mySocket);

printf(" Stisni tipko za nadaljevanje\n");

getchar();
}

// startupClient() - funkcija za prizgat socket in conektat se na server

int startupClient(unsigned short port, const char* serverName)
{
// spremenljivka za informacije o errorih

int error;

// winsock struktura tu shranimo informacije

WSAData wsaData;

// zazenemo winsock

if ((error = WSAStartup(MAKEWORD(2, 2), &wsaData)) == SOCKET_ERROR)
{
printf(" Ne morem zagnati winsock!\n");
return -1;
}

// naredimo socket

int mySocket = socket(AF_INET, SOCK_STREAM, 0);

// preverimo da ni kaj narobe

if (mySocket == SOCKET_ERROR)
{
printf(" Pri odpriranju socket nastal error!\n");
return -1;
}

struct hostent *host_entry;

// struktura z hostom in preverimo ce je server zagnan ali je sploh na internetu

if ((host_entry = gethostbyname(serverName)) == NULL)
{
printf(" Ne najdem serverja ali racunalnik ni povezan na internet!\n");
}

// strukrtura server informacij

struct sockaddr_in server;

// napolnimo socket server informacije

server.sin_family = AF_INET;
server.sin_port = htons(port);
server.sin_addr.s_addr = *(unsigned long*) host_entry->h_addr;

// se povezemo na server

if (connect(mySocket, (sockaddr*)&server, sizeof(server)) == SOCKET_ERROR)
{
printf(" Ne da se povezati na server!\n");
}

printf(" Client Prizgan Uspelo se konektat\n");

return mySocket;
}

// shutdownClient() - funkcija za ugasnit socket in izklopiti winsock

void shutdownClient(int clientSocket)

{
// zapremo socket

closesocket(clientSocket);

// ugasnemo winsock

WSACleanup();

printf(" Client Ugasnjen \n");
}
