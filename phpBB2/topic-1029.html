#include <GL\glut.h>

GLfloat xRotated, yRotated, zRotated;


void Display(void)
{
glClear(GL_COLOR_BUFFER_BIT);
glLoadIdentity();
glTranslatef(0.0,0.0,-4.0);
glRotatef(xRotated,1.0,0.0,0.0);
glRotatef(yRotated,0.0,1.0,0.0);
glRotatef(zRotated,0.0,0.0,1.0);
glScalef(2.0,1.0,1.0);
glutWireCube(1.0);
glFlush(); //Finish rendering
glutSwapBuffers();
}

void Reshape(int x, int y)
{
if (y == 0 || x == 0) return; //Nothing is visible then, so return
//Set a new projection matrix
glMatrixMode(GL_PROJECTION);
glLoadIdentity();
//Angle of view:40 degrees
//Near clipping plane distance: 0.5
//Far clipping plane distance: 20.0
gluPerspective(40.0,(GLdouble)x/(GLdouble)y,0.5,20.0);
glMatrixMode(GL_MODELVIEW);
glViewport(0,0,x,y); //Use the whole window for rendering
}

void Idle(void)
{
xRotated += 0.3;
yRotated += 0.1;
zRotated += -0.4;
Display();
}


int main (int argc, char **argv)
{
//Initialize GLUT
glutInit(&argc, argv);
glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB); //For animations you should use double buffering
glutInitWindowSize(300,300);
//Create a window with rendering context and everything else we need
glutCreateWindow("Cube example");
glPolygonMode(GL_FRONT_AND_BACK,GL_LINE);
xRotated = yRotated = zRotated = 0.0;
glClearColor(0.0,0.0,0.0,0.0);
//Assign the two used Msg-routines
glutDisplayFunc(Display);
glutReshapeFunc(Reshape);
glutIdleFunc(Idle);
//Let GLUT get the msgs
glutMainLoop();
return 0;
}

#include "GL\glut.h"

float rotationZ = 0;
float rotationEntryZ = 0;
float rotationX = 0;
float rotationEntryX = 0;
float rotationY = 0;
float rotationEntryY = 0;
void displayCB()
{
glClear(GL_COLOR_BUFFER_BIT);
glMatrixMode(GL_MODELVIEW);
glLoadIdentity();

//animation powah
glRotatef(rotationZ,0,0,1);
glRotatef(rotationX,1,0,0);
glRotatef(rotationY,0,1,0);
rotationX += rotationEntryX;
rotationY += rotationEntryY;
rotationZ += rotationEntryZ;
//lets draw something
glBegin(GL_TRIANGLES);

glColor3f(1,0,0);
glVertex2f(0,0.75);

glColor3f(0,0,1);
glVertex2f(0.75,-0.75);

glColor3f(0,1,0);
glVertex2f(-0.75,-0.75);


glEnd();

glFlush();
glutSwapBuffers();
}

// USER INTERACTION
void processSpecialKeys(int key, int x, int y) {

switch(key) {
case GLUT_KEY_LEFT :
rotationEntryZ -=0.2;
break;
case GLUT_KEY_RIGHT :
rotationEntryZ +=0.2;
break;
case GLUT_KEY_UP :
rotationEntryX -=0.2;
break;
case GLUT_KEY_DOWN :
rotationEntryX +=0.2;
break;
case GLUT_KEY_PAGE_UP :
rotationEntryY +=0.2;
break;
case GLUT_KEY_PAGE_DOWN :
rotationEntryY -=0.2;
break;
}
}

int main(int arqc, char* argv[])
{
glutInit(&arqc,argv);
glutInitDisplayMode(GLUT_DEPTH | GLUT_DOUBLE | GLUT_RGBA);
glutInitWindowPosition(100,100);
glutInitWindowSize(320,320);
int WindowID = glutCreateWindow("glTest");
glutDisplayFunc(displayCB);
glutIdleFunc(displayCB);
glutSpecialFunc(processSpecialKeys);
glClearColor(0,0,0,0);
glutMainLoop();

return 0;
}
