Ponavadi za take stvari ločeno interpoliraš kamero in tarčo. Uporabi Hermitsko al pa CatmullRom. Spline interpolacija ponavadi ne izpade preveč dobro ker ne moreš kontrolirat ekstremov. Če zveni kot kitajščina tole si boš moral pa mal matematiko pogledat al pa uporabit kakšne obstoječe knjižnice. Čeprav za tvoj konktreten problem lahko uporabiš tud čist obično linearno interpolacijo za target al pa sferično interpolacijo s kvaternioni.

ACM3D je napisal:

//-----------------------------------------------------------------------------
// Name: SetCameraPosition()
// Desc:
//-----------------------------------------------------------------------------
void CACM3D::SetCameraPosition( S_KAMERA* Kamera, BOOL Inverse )
{
//Kamera (rotiranje vektorjev)
D3DXMATRIX mtxRot,mtxSubRot1,mtxSubRot2;

D3DXMatrixRotationAxis( &mtxSubRot1, &D3DXVECTOR3(0.0f, 1.0f, 0.0f), D3DXToRadian( Kamera->AngleX ));
D3DXMatrixRotationAxis( &mtxSubRot2, &D3DXVECTOR3(1.0f, 0.0f, 0.0f), D3DXToRadian( -Kamera->AngleY ));

D3DXMatrixMultiply( &mtxRot, &mtxSubRot2, &mtxSubRot1 );

D3DXVECTOR3 Up, Look, Right;

D3DXVec3TransformCoord( &Look, &D3DXVECTOR3(0.0f, 0.0f, 1.0f), &mtxRot );
D3DXVec3TransformCoord( &Up, &D3DXVECTOR3(0.0f, 1.0f, 0.0f), &mtxRot );

D3DXVECTOR3 Position = Kamera->Position;

if(Inverse)
{
Position += Look;
Look = -Look;
}

//Izračunaj ostale vrednosti
D3DXVec3Normalize( &Look, &Look );
D3DXVec3Cross( &Right, &Up, &Look );
D3DXVec3Normalize( &Right, &Right );
D3DXVec3Cross( &Up, &Look, &Right );
D3DXVec3Normalize( &Up, &Up );

//Nastavi D3DTS_VIEW
D3DXMATRIX view;
D3DXMatrixIdentity( &view );

view._11 = Right.x;
view._12 = Up.x;
view._13 = Look.x;
view._14 = 0.0f;

view._21 = Right.y;
view._22 = Up.y;
view._23 = Look.y;
view._24 = 0.0f;

view._31 = Right.z;
view._32 = Up.z;
view._33 = Look.z;
view._34 = 0.0f;

view._41 = -D3DXVec3Dot( &Position, &Right );
view._42 = -D3DXVec3Dot( &Position, &Up );
view._43 = -D3DXVec3Dot( &Position, &Look );
view._44 = 1.0f;

m_pd3dDevice->SetTransform( D3DTS_VIEW, &view );
}

Koda:


mat4 & look_at(mat4& M, const vec3& eye, const vec3& center, const vec3& up)
{
   vec3 x, y, z;

   // make rotation matrix

   // Z vector
   z.x = eye.x - center.x;
   z.y = eye.y - center.y;
   z.z = eye.z - center.z;
   normalize(z);

   // Y vector
   y.x = up.x;
   y.y = up.y;
   y.z = up.z;

   // X =  Y cross Z
   cross(x,y,z);

   // Y = Z cross X
   cross(y,z,x);

   normalize(x);
   normalize(y);

   M.a00 = x.x; M.a01 = x.y; M.a02 = x.z; M.a03 = -x.x * eye.x - x.y * eye.y - x.z*eye.z;
   M.a10 = y.x; M.a11 = y.y; M.a12 = y.z; M.a13 = -y.x * eye.x - y.y * eye.y - y.z*eye.z;
   M.a20 = z.x; M.a21 = z.y; M.a22 = z.z; M.a23 = -z.x * eye.x - z.y * eye.y - z.z*eye.z;
   M.a30 = zero; M.a31 = zero; M.a32 = zero; M.a33 = one;
   return M;
}
