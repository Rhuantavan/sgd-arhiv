Koda:
import java.awt.*;
import java.awt.image.*;
import java.awt.event.*;

public class Game extends Frame implements Runnable, KeyListener,MouseListener,MouseMotionListener{
static final int FPS = 60;
static final int SCREEN_WIDTH = 640,
SCREEN_HEIGHT = 480,
BIT_DEPTH = 16,
REFRESH_RATE = 60;

public Game(boolean fullscreen) {
GraphicsConfiguration gc = getGraphicsConfiguration();
GraphicsDevice gd = gc.getDevice();
if(fullscreen) {
setUndecorated(true);
gd.setFullScreenWindow(this);
gd.setDisplayMode(new DisplayMode(SCREEN_WIDTH,SCREEN_HEIGHT,BIT_DEPTH,REFRESH_RATE));
}
else {

setBounds(0,0,SCREEN_WIDTH,SCREEN_HEIGHT);
setLocation(200,100);
setVisible(true);
}
createBufferStrategy(2);

addKeyListener(this);
addMouseListener(this);
addMouseMotionListener(this);

Thread t = new Thread(this);
t.start();
}

int Ix=50,Iy=50;

public void run() {
final int NANO_FRAME_LENGTH = 1000000000/FPS;
long startTime = System.nanoTime();
int frameCount = 0;
while(controls[KeyEvent.VK_ESCAPE]==0) {
BufferStrategy bs = getBufferStrategy();
Graphics g = bs.getDrawGraphics();

//preverjas kaj je pritisnjeno

if (controls[KeyEvent.VK_LEFT]==1)
{
Ix-=5;
}else
if (controls[KeyEvent.VK_RIGHT]==1)
{
Ix+=5;
}
if (controls[KeyEvent.VK_UP]==1)
{
Iy-=5;
}else
if (controls[KeyEvent.VK_DOWN]==1)
{
Iy+=5;
}



//Izrisovanje
g.setColor(Color.white);
g.fillRect(0,0,SCREEN_WIDTH,SCREEN_HEIGHT);

g.setColor(Color.blue);
g.fillOval(Ix-10,Iy-10,20,20);



bs.show();
frameCount++;

while((System.nanoTime()-startTime)/NANO_FRAME_LENGTH <frameCount) {
Thread.yield();
}
}

dispose();
}

int [] controls = new int[256];

public void keyPressed(KeyEvent ke) {
controls[ke.getKeyCode()] = 1; //key handlers should only set flags. The interpretation of the flags is best left to the game loop
}

public void keyReleased(KeyEvent ke) {
controls[ke.getKeyCode()] = 0; //key handlers should only set flags. The interpretation of the flags is best left to the game loop
} public void keyTyped(KeyEvent ke) {}

public void mouseClicked(MouseEvent e)
{
}
public void mousePressed(MouseEvent e)
{
}
public void mouseReleased(MouseEvent e)
{
}
public void mouseEntered(MouseEvent e)
{
}
public void mouseExited(MouseEvent e)
{
}
public void mouseDragged(MouseEvent e)
{
}
public void mouseMoved(MouseEvent e)
{
Ix=e.getX() ;
Iy=e.getY();
}

public static void main(String [] args) {
new Game(false);
}
}



V metodi run je glavna zanka. V njej najprej preverjaš kaj je bilo pritisnjeno ipd potem glede na to premikaš objekte in na koncu izrišeš vse kar je izrisat treba.

while(nitka == thisThread)
{
zX = (int)tmpX;
zY = (int)tmpY;
if( zX == kX && zY == kY )
{
premik = false; //po koncanem premiku nastavimo premik na false
