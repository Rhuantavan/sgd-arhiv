// funkcija obrne stevilo
// z uporabo nizov
int obrniStevilo(int stevilo)
{

// uporabimo dva pomozna niza, vec kot 255 cifer verjetno ne bomo imeli v praksi...
char strZacetnoStevilo[255]; // predstavitev prvega stevila z nizom
char strObrnjenoStevilo[255]; // predstavitev obrnjenjega stevila z nizom


_itoa(stevilo,strZacetnoStevilo,10); // pretvorimo stevilo v niz

int stStevk = strlen(strZacetnoStevilo) - 1; // preberemo koliko cifer je v nizu

// obracamo nize
for (int i=0; i<=stStevk; i++)
strObrnjenoStevilo[stStevk-i] = strZacetnoStevilo[i]; // prepisemo cifro iz meta i v prvem nizu na mesto dolzina-i v drugem nizu

// drugi niz moramo sedaj se zakljuciti z vrednostjo 0
strObrnjenoStevilo[stStevk+1] = 0;

//sedaj pa obrnjen niz pretvorimo v stevilo in vrnemo to stevilo
return atoi(strObrnjenoStevilo);

}


void main()
{
int stevilo, obrnjenoStevilo;

cin >> stevilo;

Evo, ker se mi je dalo, sem naredil še verzijo brez nizov. Mogoče za začetek malce o algoritmu. Enomestna števila ostanejo enaka zato jih lahko kar vrnemo, medtem, ko moramo za večmestna števila najprej najti prvo večje število z vodilno 1 in sledečimi ničlami (10,100,1000). Ko najdemo tako število poznamo pomen vsake cifre števila. Če je število recimo 543, je prvo iskano večje stevilo 1000. Sedaj vemo, da prva cifra, torej 5 pomeni stotice v našem številu. Ker bo v obrnjenem številu ta cifra na zadnjem mestu bo tam pomenila enice. Pomen vsake nalsednje cifre v originalnem številu pada, v obrnjenem pa raste. S pomočjo teh dveh količnikov pa že lahko opravimo obračanje. Tule je primerček na katerem bo vse bolj jasno:









Koda:

543 - prvo vecje stevilo je 100

Stevilo Obrnjeno stevilo
Stevilo Kolicnik Cifra Stevilo Kolicnik Cifra
543 - 100 * 5 = 0 + 1 * 5 =
43 - 10 * 4 = 5 + 10 * 4 =
3 - 1 * 3 = 45 + 100 * 3 =
0 345




Pa še koda funkcije. Tokrat z ziheraškimi dodatki: negativna števila ohranjajo predznak (-543 se obrne v -345), hitro vračanje števil med -9 in 9, ki ostanejo enaka...








Koda:

// funkcija obrne stevilo
// brez uporabe nizov
int obrniStevilo2(int stevilo)
{

bool negativno = false; // zastavica s katero si oznacimo ali je bilo osnovno stevilo negativno
unsigned char stCifer = 0; // st stevk v osnovnem stevilu
int kolicnik = 1; // kolicnik, ki ga uporabljamo za racunanje trenutne cifre osnovnega stevila
int kolicnikObr = 1; // kolicnik, ki ga uporabljamo za racunanje vrednosti, ko jo moramo dodati obrnjenemu stevilu
unsigned cifra; // trenutna cifra
int obrnjenoStevilo = 0; // obrnjeno stevilo, ki ga bomo vrnili


// najprej preverimo ali je stevilo 0
if (stevilo == 0)
return stevilo; // potem je tudi rezultat 0
// preverimo ali je stevilo negativno
if (stevilo < 0)
{
negativno = true; // oznacimo si, da je stevilo negativno
stevilo = -stevilo; // vzamemo absolutno vrednost stevila saj bomo obracali pozitivno stevilo nato pa nazaj dodali predznak
}


// najprej moramo dolociti koliko cifer ima stveilo
// to bomo naredili tako, da najdemo prvo vecje stevilo z zacetno enico in nato niclam, torej 1,10,100,1000..
// ce vzamemo za primer stevilo 543 potem bo prvo vecje stevilo 1000
while (kolicnik < stevilo)
{
stCifer++; // oznacimo si da ima stevilo eno cifro vec
kolicnik *= 10; // kolicnik povecamo za 10
}


// iskali smo prvo vecje stevilo v nadaljevanju pa rabimo dostop do vrednosti cifre zato moramo zmanjsati kolicnik
// recimo ce je bilo osnovno stevilo 544, je bilo prvo vecje stevilo 1000. sedaj vemo, da prva cifra predstavlja stotice
// zato kolicnik postavimo na 100
kolicnik /= 10; // kolicnik zmanjsamo

// obdelali bomo vse cifre osnovnega stevila
for (int i=0; i<stCifer; i++)
{
// najprej izracunamo trenutno cifro npr. 543/100 = 5, 43/10 = 4, 3/1 = 3
cifra = stevilo/kolicnik;

// ker bomo to cifro ze dali v obrnjeno stevilo je v osnovnem stevilu vec ne rabimo
// vrednost, ki jo moramo odsteti dobimo iz osnovnega kolicnika ter cifre
// primer 543 - 5*100 = 43, 43 - 4*10 = 4, 3-3*1 = 0;
stevilo = stevilo - cifra * kolicnik;

// cifro moramo dodati v obrnjeno stevilo
// tukaj je pomen cifre ravno obraten, ce je bila cifra prej na prvem mestu potem sedaj predstavlja enice,
// ce je bila na drugem potem predstavlja desetice...
// zato uporabljamo dodaten kolicnik, ki se v vsaki zanki poveca * 10. torej 1, 10, 100 , 1000...
// trenutno cifro zmnozimo s tem kolicnikom in dobimo vrednost, ki jo moramo pristeti obrnjeni stevilki
// primer 0 + 5*1 =5, 5 + 4*10 = 45, 45 * 3*100 = 345
obrnjenoStevilo = obrnjenoStevilo + cifra * kolicnikObr;

// kolicnik s katerim dolocamo pomen mest v osnovnem stevilu moramo sedaj pomanjsati na naslednje mesto
kolicnik /= 10;

// kolicnik s katerim dolocamo pomen mest v obrnjenem stevilu pa je potrebno povecati, saj bo nova cifra
// prisla na mesto pred vse ostale
kolicnikObr *= 10;
}


if (negativno) // ce je bilo zacetno stevilo negativno
obrnjenoStevilo = -obrnjenoStevilo; // potem naj bo tudi obrnjeno stevilo negativno

return obrnjenoStevilo; // vrnemo vrednost obrnjenega stevila

}
