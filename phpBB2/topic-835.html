No koda je takala (cela metoda):









Koda:

private void BuildLevel ( string LevelFilename )
{
// TO-DO: add decription of the level file here (once encription is implemented

XmlDocument levelDocument = new XmlDocument();
levelDocument.Load (LevelFilename); // load the level XML file

#region Level Info
XmlNode levelNode = levelDocument.SelectSingleNode ( "level" );
levelInfo = new LevelInfo ();
levelInfo.Name = levelNode.Attributes.GetNamedItem("name").InnerText;
levelInfo.Author = levelNode.Attributes.GetNamedItem("author").InnerText;
#endregion // Level Info

#region Paddles
paddles = new PaddleCollection (); // make a new instance of the paddle collection (could use generics here)
XmlNodeList paddleNodes = levelDocument.ChildNodes;
foreach ( XmlNode paddleNode in paddleNodes)
{
Paddle tempPaddle;

// get paddle parameteres from the XML node <paddle>
int left = Int32.Parse ( paddleNode.Attributes.GetNamedItem ( "left" ).InnerText );
int top = Int32.Parse ( paddleNode.Attributes.GetNamedItem ( "top" ).InnerText );
int width = Int32.Parse ( paddleNode.Attributes.GetNamedItem ( "width" ).InnerText );
int height = Int32.Parse ( paddleNode.Attributes.GetNamedItem ( "height" ).InnerText );

// object can have a texture or a color
if ( paddleNode.Attributes.GetNamedItem ( "color" ).InnerText == "" )
{
Bitmap texture = new Bitmap ( Application.StartupPath + paddleNode.Attributes.GetNamedItem ( "texture" ) );
tempPaddle = new Paddle ( left, top, width, height, texture );
}
else
{
string colorAtribute = paddleNode.Attributes.GetNamedItem ( "color" ).InnerText;

int r = Int32.Parse ( colorAtribute.Substring ( 0, 3 ) );
int g = Int32.Parse ( colorAtribute.Substring ( 4, 3 ) );
int b = Int32.Parse ( colorAtribute.Substring ( 8, 3 ) );
Color color = Color.FromArgb ( r, g, b );
tempPaddle = new Paddle ( left, top, width, height, color );
}
paddles.Add ( tempPaddle ); // finnaly add the new paddle to the collection
}
#endregion // Paddles

}
