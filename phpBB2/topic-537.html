   UdelezenecProjekta *konec_u;
   zacetek_u->ime="testni";

   zacetek_u->naslov.ulica="ulica";

   zacetek_p->udelezenci=zacetek_u;
   zacetek_p->konec=zacetek_u;
   konec_u=zacetek_u;

   izpisiMeni(zacetek_p,konec_p);
   pobrisiUdelezence(zacetek_p);
   pobrisiProjekte(zacetek_p);
   return 0;
}
