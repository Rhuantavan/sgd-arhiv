Tole je v SaveGame classu
Za vsak chunk posebaj.








Koda:
public static void saveBackBlocks(World world, int chunkX, int chunkY)
{
string documents = Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments);
string activeDir = System.IO.Path.Combine(documents,"Castler/Save/back");
if (System.IO.Directory.Exists(activeDir) != true)
{
System.IO.Directory.CreateDirectory(activeDir);
}
string saveString = "";

for (int y = 0; y < 50; y++)
{
for (int x = 0; x < 50; x++)
{
if (x == 50 - 1)
{
saveString += world.backBlocksId[y + chunkY * 50, x + chunkX * 50] + "" + Environment.NewLine;
}
else
{
saveString += world.backBlocksId[y + chunkY * 50, x + chunkX * 50] + ".";
}
}
}
System.IO.File.WriteAllText(activeDir + @"\backBlocks." + chunkX + "." + chunkY + ".txt", saveString);

}


V World classu je pa zanka skozi katero grejo vsi chunki
