ne dela zato, ker moraš "exportat" class (kličeš v bistvu this.text()).

ker pa class za razliko od funkcije porabi nekaj pomnilnika, moraš le tega najprej naresti/alocirat na DLL strani (vsak dll ima svoj memory), vrnit ta pointer do uporabnika in potem izbrisat class (zopet na na dll strani).

osnovna varjanta je z uporabo dveh funkcij:

moj_class* Create() { return new moj_class;}
void Free(moj_class* lPtr) { delete lPtr;}

obe definiraš v dll headerju, in jih potem uporbljaš v svoji kodi,(getprocaddress).

moj_class* p1 = (getprocadress)Create();
p1->text();
(getprocaddress) Free(p1);

druga, boljša varjanta je pa uporaba kwazi COM-a in smart pointerjev...

vsak objekt ima števec referenc (kolikokrat je ta objekt uporabljen). ko se objekt kreira je to število 0. zraven imaš tudi dve funkciji AddRef() (poveča števec za ena) in pa Release() (zmanjša števec za ena, if števec = 0, delete this).

CBase* ptr = new CBase;
ptr->AddRef(); dodamo referenco
ptr->text(); uporabimo
ptr->Release(); izbrišemo, ne rabimo klicat delete ptr!!!


no, smart pointerji pa vso zadevo avtomatizirajo
void main()
{
smart_ptr<CBase> ptr = new CBase;
ptr->text();
}

ko smart_ptr "pade" ven iz funkcije/scope-a, se avtomatsko izbiše tudi alocirani objekt...

Sesuje se točno za tistim, ko pokliče constructor:

V klientu








Koda:
A* a = (pfnCreateA1)();



V DLL








Koda:
printf("In Default Constructor of A in DLL1. m_nNum = %d\n", m_nNum);



Konstruktor se izvede normalno, problem se pojavi, ko hoče klient poklicati metodo SetNum(int):









Koda:
a->SetNum(1);

, ki je takoj za klicem konstruktorja.

delete a; pride na vrsto šele pozneje.

Celoten klient:








Koda:
#include "stdafx.h"
#include "expclass.h"
#include <malloc.h>
#include <windows.h>

int main(int argc, char* argv[])
{

TCHAR szDllPath[MAX_PATH];
HMODULE hMod;

do
{
_tprintf(TEXT("Please enter the dll to load the class from:\n"));
_tscanf(TEXT("%s"), szDllPath);
}
while ((hMod = LoadLibrary(szDllPath)) ?
false :
_tprintf(TEXT("Failed to Load the dll. GetLastError returns: 0x%X\n"), GetLastError()));


/* Virtual Function table method */
_tprintf(TEXT("Importing by Virtual Function Table:\n\n"));

typedef A* (*PFNCreateA1)();
typedef A* (*PFNCreateA2)(int);

/*************************************************************************/
_tprintf(TEXT("Creating an Object of A through Default Constructor\n"));

PFNCreateA1 pfnCreateA1 = (PFNCreateA1)GetProcAddress(hMod, TEXT("CreateObjectofA1"));

A* a = (pfnCreateA1)(); // še dela

_tprintf(TEXT("Value of m_nNum in a is %d\n"), a->GetNum());

delete a;

a = NULL;
/*************************************************************************/
_tprintf(TEXT("Creating an Object of A through Other Constructor\n"));

PFNCreateA2 pfnCreateA2 = (PFNCreateA2)GetProcAddress(hMod, TEXT("CreateObjectofA2"));

a = (pfnCreateA2)(2);

_tprintf(TEXT("Value of m_nNum in a is %d\n"), a->GetNum());

_tprintf(TEXT("Value of m_nNum in a is %d\n"), a->GetNum());

delete a;

/*************************************************************************/



return 0;
}

// header
class IRefCounted
{
public:
virtual void ~IRefCounted() {} = 0; // pure virtua
void AddRef() {++mRefCounter;}
void Release() {--mRefCounter; if (mRefCounter <= 0) delete this;}
private:
int mRefCounter; // = 0 v construktorju
};

extern "C"
{
IRefCounted* Create(); // prototip funkcije
}
