#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <netdb.h>
#include <sys/types.h>
#include <netinet/in.h>
#include <sys/socket.h>

#define PORT 6666 // na ker port se bomo konektali
#define MAXDATASIZE 100 // maximalno stevilo bajtov ki jih lahko dobimo naenkrat

int main(int argc, char *argv[])
{
        int sockfd, numbytes;
        char buf[MAXDATASIZE];
        struct hostent *he;
        struct sockaddr_in their_addr;  // od servera ip
  if (argc != 2)
        {
                fprintf(stderr, "usage: 127.0.0.1 \n ");
                exit(1);
        }

        if ((he=gethostbyname(argv[1])) == NULL) // dobim host informacije
        {
                perror("gethostbyname");
                exit(1);
        }

        if ((sockfd = socket(AF_INET, SOCK_STREAM, 0)) == -1)
        {
                perror("socket");
                exit(1);
        }

        their_addr.sin_family = AF_INET; // protokol
        their_addr.sin_port = htons(PORT); //na kater port se konektamo
        their_addr.sin_addr = *((struct in_addr *)he->h_addr);  <-- tu bi po moje moral bit ip oz. host od servera
        memset(&(their_addr.sin_zero), '\0', 8);

if (connect(sockfd, (struct sockaddr *)&their_addr, sizeof( struct sockaddr)) == -1);
        {
                perror("connect");
                exit(1);
        }

        if ((numbytes=recv(sockfd, buf, MAXDATASIZE -1, 0)) == -1);
        {
                perror("recv");
                exit(1);
        }

        buf[numbytes] = '\0';

        printf(" Sprejem Bajtov: %s, buf");

        close(sockfd);

        return 0;
}
