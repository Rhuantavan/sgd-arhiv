Če prav razumem gre za in game menuje. Obstaja tudi možnost da uporabiš windows kontrole, ampak to ponavadi izpade neprofesionalno. Dosti bolje izpadejo custom made menuji ki jih sam rišeš in nadziraš njihovo funkcionalnost. Najboljši pristop je verjetno da narediš en class za izbiro v menuju recimo (vse grafične elemente lahko tudi recimo implementaš iz skupnega interfacea, da imaš lepšo kodo navzven), ki kontrolira klikanje, evente, izris. Kot da bi naredil custom kontrolo v winsih pravzaprav, le da malo več stvari nadziraš.

Preprosto sicer je bolj z imageboxi, vendar ko bo projekt rasel boš imel na enem formu že toliko imageboxov da bo stvar postala nepregledna in neuporabna za debugiranje (govorim iz lastnih izkušenj).

Public Class GUIButton
    Implements IGUIPositionableElement
    Private mTrigger As Integer
    Private mRect As Rectangle
    Private mPosX, mPosY As Integer
    Private mName As String
    Private mText As String
    Private mMouseOver As Boolean
    Private mVisible As Boolean
    Private mContext As Object
    Private mEnabled As Boolean
    Private parent As ITriggerHost

    Public Property Context() As Object
        Get
            Return mContext
        End Get
        Set(ByVal Value As Object)
            mContext = Value
        End Set
    End Property

    Public Property Text() As String
        Get
            Return mText
        End Get
        Set(ByVal Value As String)
            mText = Value
            Dim bmp As New Bitmap(1, 1)
            Dim g As Graphics = Graphics.FromImage(bmp)
            Dim layout As New SizeF(wid - CInt(mPosX * wid / 800), hei - CInt(mPosY * hei / 600))
            Dim s As SizeF = g.MeasureString(mText, MeasureFont, layout, StringFormat.GenericDefault)
            mRect = New Rectangle(CInt(mPosX * wid / 800), CInt(mPosY * hei / 600), CInt(s.Width), CInt(s.Height))
            g.Dispose()
            bmp.Dispose()
        End Set
    End Property

    Public Property Enabled() As Boolean Implements IGUIElement.Enabled
        Get
            Return mEnabled
        End Get
        Set(ByVal Value As Boolean)
            mEnabled = Value
        End Set
    End Property

    Public Sub Reposition(ByVal posx As Integer, ByVal posy As Integer)
        mPosX = posx
        mPosY = posy
        mRect.X = posx
        mRect.Y = posy
    End Sub

    Public ReadOnly Property Rect() As Rectangle Implements IGUIPositionableElement.Rect
        Get
            Return mRect
        End Get
    End Property

    Public Sub New(ByVal name As String)
        mName = name
        mVisible = True
        mEnabled = True
    End Sub

    Public Sub New(ByVal name As String, ByVal par As ITriggerHost, ByVal trigger As Integer, ByVal text As String, ByVal posx As Integer, ByVal posy As Integer, Optional ByVal context As Object = Nothing)
        Me.New(name)
        parent = par
        mTrigger = trigger
        mText = text
        mPosX = posx
        mPosY = posy
        mContext = context
        Dim bmp As New Bitmap(1, 1)
        Dim g As Graphics = Graphics.FromImage(bmp)
        Dim layout As New SizeF(wid - CInt(mPosX * wid / 800), hei - CInt(mPosY * hei / 600))
        Dim s As SizeF = g.MeasureString(mText, MeasureFont, layout, StringFormat.GenericDefault)
        mRect = New Rectangle(CInt(mPosX * wid / 800), CInt(mPosY * hei / 600), CInt(s.Width), CInt(s.Height))
        g.Dispose()
        bmp.Dispose()
    End Sub

    Public ReadOnly Property MouseOver() As Boolean
        Get
            Return mMouseOver
        End Get
    End Property

    Public Sub Process() Implements IGUIElement.Process
        If mVisible = False Then Exit Sub
        If mEnabled AndAlso mouseX >= mRect.Left AndAlso mouseX <= mRect.Right AndAlso mouseY >= mRect.Top AndAlso mouseY <= mRect.Bottom Then
            mMouseOver = True
            If prevMouseB1 = False And mouseB1 = True Then
                parent.Trigger(Me, mTrigger, mContext)
            End If
        Else
            mMouseOver = False
        End If
    End Sub

    Public Sub Draw() Implements IGUIElement.Draw
        If mVisible = False Then Exit Sub
        Dim badRect As Rectangle = mRect
        badRect.Width += badRect.X
        badRect.Height += badRect.Y
        If mEnabled Then
            If mMouseOver Then
                GUIFont.DrawText(mText, badRect, DrawTextFormat.WordBreak, Color.Red)
            Else
                GUIFont.DrawText(mText, badRect, DrawTextFormat.WordBreak, Color.White)
            End If
        Else
            GUIFont.DrawText(mText, badRect, DrawTextFormat.WordBreak, Color.Gray)
        End If
    End Sub

    Public Sub Load(ByVal sr As System.IO.StreamReader) Implements IGUIElement.Load
        'trigger
        'text
        'posX800x600
        'posY800x600
        Dim nf As New NumberFormatInfo()
        mTrigger = Integer.Parse(sr.ReadLine.Trim, nf)
        mText = sr.ReadLine.Trim
        mPosX = Integer.Parse(sr.ReadLine.Trim, nf)
        mPosY = Integer.Parse(sr.ReadLine.Trim, nf)
        Dim bmp As New Bitmap(1, 1)
        Dim g As Graphics = Graphics.FromImage(bmp)
        Dim layout As New SizeF(wid - CInt(mPosX * wid / 800), hei - CInt(mPosY * hei / 600))
        Dim s As SizeF = g.MeasureString(mText, MeasureFont, layout, StringFormat.GenericDefault)
        mRect = New Rectangle(CInt(mPosX * wid / 800), CInt(mPosY * hei / 600), CInt(s.Width), CInt(s.Height))
        g.Dispose()
        bmp.Dispose()
        'GUIFont.DrawText(mText, mRect, DrawTextFormat.CalculateRect, Color.Red)
    End Sub

    Public ReadOnly Property Name() As String Implements IGUIElement.Name
        Get
            Return mName
        End Get
    End Property

    Public Property Visible() As Boolean Implements IGUIElement.Visible
        Get
            Return mVisible
        End Get
        Set(ByVal Value As Boolean)
            mVisible = Value
        End Set
    End Property

    Public Property X() As Integer Implements IGUIPositionableElement.X
        Get
            Return mPosX
        End Get
        Set(ByVal Value As Integer)
            Me.Reposition(Value, mPosY)
        End Set
    End Property

    Public Property Y() As Integer Implements IGUIPositionableElement.Y
        Get
            Return mPosY
        End Get
        Set(ByVal Value As Integer)
            Me.Reposition(mPosX, Value)
        End Set
    End Property
End Class
