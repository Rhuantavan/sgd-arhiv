#include <afxwin.h>  // mores imet ta glavni fajl ce delas v mfcju
#include "intro.h"  // header fajl

CMainWin::CMainWin()
{
   // tukaj pride vsa inicializacija programa

   RECT r; // to bomo uporabljali da povemo velikost okna

   r.left = 300; // top in left se uporablja da vemo kje se bo okno zacelo izrisovat
   r.top = 300;  // in right in bottom kje se bo koncalo
   r.right = 600;
   r.bottom = 600;

    Create(NULL, " Created by Mafia Software ", WS_OVERLAPPEDWINDOW, r);

    BOOL CApp::InitInstance()
   {
      // InitInstance je virtualna funkcijaki more bit "overridden" od mojega app classa
      // To klicemo vsakic ko se aplikacija zazene
      // m_pMainWnd je pointer do mojega glavnega okenskega objekta
      // m_pMainWnd->ShowWindow(m_nCmdShow) pove kako se bo program izrisal

      m_pMainWnd = new CMainWin;
      m_pMainWnd->ShowWindow(m_nCmdShow);
      m_pMainWnd->UpdateWindow();

      return TRUE;
   }

   // message map je uporabljeno da "procesing" messages ki jih OS posilja mojemu programu
   // Primer: Ce program maximiras oz. minimiras bo okno potrebovalo da bo prebarvano
   // dokler okno ne ve kako se prepleskat je moja dolznost kot programer da poskrbim za ta "task"
   // Ce nimas nicesa da bi se odzvalo na paint message, potem ce kako drugo okno gre cez moje ...
   // v tem prog. se samo hocemo odzvati na paint message...

   BEGIN_MESSAGE_MAP (CMainWin, CFrameWnd)
      ON_WM_PAINT() // hocemo da se prog. odzove na paint msg
    END_MESSAGE_MAP()

   // tu je koda ki jo bo program izpeljav ce sprejme paint msg, ko se zazene
   // sprejme paint msg in se vse lepo izpise

   afx_msg void CMainWin::OnPaint()
   {
      CPaintDC dc(this); // to je edino uporabljeno z paint msg

      char str[25]; // to bo shranilo moje ime

      // wsprintf ali sprintf so uporabljeni da kopira string na desni v stevilo na levi
      // obojno je podobno printf ker lahko kopiras premenljivke v string
      // primer:
      // int denar;
      // denar = 50;
      // wsprintf(str, "you owe me %d dollars.", denar);

      // sedaj kopirajmo moje ime v array oz. polje
      wsprintf(std, " Enostavna MFC aplikacija ");

      // izpise moje ime na (100 (x kordinati), 125 ( y kordinati)
      // dc je "device context" objekt, in TextOut je funkcija, ki
      // je vkljucena v CPaintDC razredu.

      dc.TextOut(80, 125, str, strlen(str));
   }

   CApp MafiaSoftware; // to naredi aplikacisko objekt
}
