DODATEK

Članku so priložene tudi datoteke, v katerih imate celotne programe, ki smo jih skozi članek sestavljali košček za koščkom. Kot bonus, ki naj še dodatno pripomore k razumevanju sporočil, pa podajmo še en primer kode. V njej se zraven že omenjenih sporočil, odzivamo tudi na pritiske tipk. V primeru, da uporabnik pritisne katero od alfanumeričnih tipk, prikažemo sporočilno okno, ki izpiše zadnjo tipko. Če uporabnik pritisne tipko Escape, pa prikažemo sporočilno okno, ki uporabnika vpraša ali želi končati delo s programom. Prav tako je v kodi dodanih nekaj varnostnih mehanizmov, ki preverijo ali so se funkcije izvedle uspešno ter ob neuspehu opozorijo na napako.

#include <windows.h>


// deklariramo našo okensko proceduro
// implementacijo bomo napisali kasneje tukaj samo povemo da obstaja
long CALLBACK MyWndProc(HWND hWnd, UINT uMessage, WPARAM wParam, LPARAM lParam);


int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int cmdShow)
{
  // spremenljivka v katero si bomo shranili ročaj našega okna
  HWND hWnd;

  // struktura s katero bomo določili lastnosti našega razreda oken
  WNDCLASSEX wc;

   // določimo velikost strukture
  wc.cbSize = sizeof(WNDCLASSEX);
  // nastavimo obnašanje okna
  wc.style = CS_HREDRAW | CS_VREDRAW | CS_OWNDC; 
  // naslednja 2 parametra sta za dodatne informacije o stilu okna, ki jih
  // ne potrebujemo, zato oba nastavimo na 0 (ignoriramo)
  wc.cbClsExtra = 0;                
  wc.cbWndExtra = 0;              
  // nastavimo okensko proceduro, ki bo skrbela za obdelovanje sporočil
  wc.lpfnWndProc = MyWndProc;   
  // nastavimo ročaj aplikacije, ki je lastnik tega razreda oken
  wc.hInstance = hInstance;
  // določimo barvo ozadja našega okna
  wc.hbrBackground = (HBRUSH)GetStockObject(DKGRAY_BRUSH); 
  // ikona okna v normalnem stanju
  wc.hIcon = LoadIcon(NULL, IDI_APPLICATION);  
  // ikona okna v pomanjšanem stanju
  wc.hIconSm = LoadIcon(NULL, IDI_APPLICATION);
  // miškin kazalec, ki ga želimo uporabljati
  wc.hCursor = LoadCursor(NULL, IDC_ARROW); 
  // ime menija, ki ga naj ima naše okno
  wc.lpszMenuName = NULL;           
   // našemu razredu oken določimo ime
  wc.lpszClassName = "MojRazredOken";   
 
  // registracija našega razreda oken
  RegisterClassEx(&wc);

  // ustvarimo novo okno
  hWnd = CreateWindowEx(
           // razširjen still okna
           NULL,
           // ime razreda oken, ki predstavlja osnovo tega okna
          "MojRazredOken",
           // ime okna (besedilo, ki se prikaže v naslovni vrstici)         
           "Moje prvo okno",           
           // stil okna
           WS_OVERLAPPEDWINDOW,  
           // x koordinata levega roba okna - uporabimo privzeto vrednost
           CW_USEDEFAULT,
           //y koordinata zgornjega roba okna  - uporabimo privzeto vrednost
           CW_USEDEFAULT,        
           // širina okna (v pikslih)
           512,                   
           // višina okna (v pikslih)
           512,                  
           // ročaj starševsekga okna
           NULL,
           // ročaj menija
           NULL,                 
           // ročaj aplikacije, ki je lastnik okna
           hInstance,            
           // kazalec na dodatne informacije
           NULL);   

  // preverimo ali je bilo ustvarjanje okna uspešno
  if (hWnd == NULL)
  {
     // če je kazalec na novo okno NULL je bilo ustvarjanje neuspešno
     // zato najprej prikažemo sporočilno okno, ki javi napako
     MessageBox(NULL,"Napaka pri kreiranju okna", "NAPAKA",
                MB_ICONSTOP | MB_OK);

    // nato pa prekinemo naš program (ne dela brez okna ;-) )
     return –1;             
  }

  // operacijskemu sistemu sporočimo naj prikaze okno
  ShowWindow(hWnd, cmdShow); 
   // operacijskemu sistemu sporočimo naj izriše vsebino okna
  UpdateWindow(hWnd);
  // spremenljivka v katero shranimo trenutno sporočilo
  MSG msg;

  // poženemo zanko s sporočili
  // zanka se izvaja dokler se v vrsti ne pojavi sporočilo Quit(WM_QUIT)
  while ( GetMessage(&msg, NULL, 0, 0))
  {
    // pretvorimo sporočilo v primernejši format
    // virtualne kode tipk pretvorimo v ASCII znake
    TranslateMessage(&msg);

    // sporočilo pošljemo glavni okenski proceduri, da ga obdela
    DispatchMessage(&msg);
  }

  // končamo izvajanje programa in vrnemo kot rezultat vrednost
  // prvega parametra zadnjega sporočila
  return msg.wParam;
}


long CALLBACK MyWndProc(HWND hWnd, UINT uMessage, WPARAM wParam, LPARAM lParam)
{
  HDC hDC;
  PAINTSTRUCT PaintStruct;

  switch (uMessage)
  {
    // WM_CREATE - sporočilo se sproži ob kreiranju okna
    {
      // izvedemo inicializacijo, ki je potrebna ob kreiranju
      return 0;
    }

    // WM_DESTROY - sporočilo označuje, da želi uporabnik zapreti okno
    case WM_DESTROY:
    {
      // operacijskemu sistemu pošljemo sporočilo WM_QUIT.
      PostQuitMessage(0);
      return 0;
    }

    // WM_PAINT - sporočilo pove, da je treba osvežiti vsebino okna
    case WM_PAINT:
    {
      // pripravimo okno za risanje
      hDC = BeginPaint(hWnd,&PaintStruct);

      // rišemo  

      // obvestimo okno, da smo končali z risanjem
      EndPaint(hWnd,&PaintStruct);
      return 0;
    }

    // WM_CHAR - sporočilo pove, da je uporabnik vnesel nek znak
    //(pritisnil in spustil tipko)
    case WM_CHAR:
    {
      // ASCII koda pritisnjene tipke se shrani v parameter wParam.
      // preverimo ali je pritisnjena tipka kateri od alfanum. znakov

      if (wParam > 32 && wParam <128)       
      {
        // pripravimo si niz, ki ga želimo izpisati v sporočilnem oknu
        char strNizTipka[25] = "Pritisnili ste tipko ' '";

        // v niz na mesto med narekovaji vpišemo znak,
        // ki ga je uporabnik vnsesel
        strNizTipka[22] = char(wParam);

        // prikažemo sporočilno okno, ki izpiše pripravljen niz
        MessageBox(hWnd, strNizTipka,"Tipka", MB_OK | MB_ICONINFORMATION);
      }

      return 0;
    }

    // WM_KEYDOWN - sporočilo pove da je neka tipka pritisnjena
    // (uporabnik drži tipko)
    // sporočilo uporabljamo za preverjanje stanja tipk, ki ne vrnejo znaka
    // (torej ne pride do sporocila WM_CHAR)
    // take tipke so recimo F1.. F12, Esc, Home, End...
    case WM_KEYDOWN:
    {
      // te tipke imajo virtualno kodo (virtual key code),
      // ki se shrani v parameter wParam
      switch(wParam)
      {
        // če je uporabnik pritisnil tipko Esc
        case VK_ESCAPE:    
        {
          // spremenljivka v katero si shranimo odgovor sporočilnega okna
          int najKoncam;

          // prikažemo sporočilno okno v katerem uporabnika vprašamo
          // ali želi končati aplikacijo
          najKoncam = MessageBox(hWnd,"Naj zaprem aplikacijo ?", "Izhod",
                                 MB_OKCANCEL | MB_ICONQUESTION);

          // če je uporabnik pritisnil gumb OK potem želi
          // končati aplikacijo
          if (najKoncam == IDOK)
          {
            // zato pošljemo v sporočilno vrsto sporocilo WM_QUIT
            PostQuitMessage(0);
            // in končamo izvajanje okenske procedure
            return 0;
          }

          // če je uporabnik pritisnil gumb Cancel
          else
           // pa samo prekinemo odločitveno zanko
           break;
        }
        // če je pritisnjena katerakoli druga tipka končamo preverjanje
        default:
          break;
      }
      return 0; // ker smo obdelali sporočilo končamo okensko proceduro
    }

    default:
    // neobdelana sporočila pošljemo privzeti okenski proceduri
    return DefWindowProc(hWnd,uMessage,wParam, lParam);
  }

}
